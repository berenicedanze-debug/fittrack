        function handleSaveWorkout(e) {
            e.preventDefault();
            const disc = document.getElementById('form-discipline').value;
            const newWorkout = {
                id: Date.now().toString(),
                discipline: disc,
                date: document.getElementById('form-date').value,
                duration: parseInt(document.getElementById('form-duration').value),
                calories: parseInt(document.getElementById('form-calories').value || 0),
                feeling: document.getElementById('form-feeling').value,
                notes: document.getElementById('form-notes').value
            };

            if (disc === 'Course') {
                newWorkout.distance = document.getElementById('form-distance')?.value;
                newWorkout.pace = document.getElementById('form-pace')?.value;
                newWorkout.elevation = document.getElementById('form-elevation')?.value;
            } else if (disc === 'Natation') {
                newWorkout.distance = document.getElementById('form-distance')?.value;
                newWorkout.laps = document.getElementById('form-laps')?.value;
                newWorkout.stroke = document.getElementById('form-stroke')?.value;
            } else if (disc === 'Tennis') {
                newWorkout.matchScore = document.getElementById('form-score')?.value;
                newWorkout.intensity = document.getElementById('form-intensity')?.value;
            } else if (disc === 'Pilates') {
                newWorkout.focusArea = document.getElementById('form-focus')?.value;
                newWorkout.intensity = document.getElementById('form-intensity')?.value;
            } else if (disc === 'Musculation') {
                newWorkout.exercises = document.getElementById('form-exercises')?.value;
                newWorkout.setsReps = document.getElementById('form-setsreps')?.value;
            }

            workouts.unshift(newWorkout);
            saveState();
            closeLogModal();
            renderDashboard();
            renderLogs();
            renderCalendar();
        }

            if(confirm("Supprimer cette séance ?")) {

                saveState();
                renderDashboard();

                renderCalendar();
            }
        }

        // ================= CALENDAR & SCHEDULE CONTROLLER =================
        function renderCalendar() {
            const year = currentCalendarDate.getFullYear();
            const month = currentCalendarDate.getMonth();

            const monthNames = ["Janvier", "Février", "Mars", "Avril", "Mai", "Juin", "Juillet", "Août", "Septembre", "Octobre", "Novembre", "Décembre"];
            document.getElementById('calendar-month-year').innerText = `${monthNames[month]} ${year}`;

            const grid = document.getElementById('calendar-grid');
            grid.innerHTML = '';

            const firstDayIndex = new Date(year, month, 1).getDay() || 7; // Monday = 1
            const totalDays = new Date(year, month + 1, 0).getDate();

            // Empty cells for alignment
            for (let i = 1; i < firstDayIndex; i++) {
                const emptyCell = document.createElement('div');
                emptyCell.className = "h-20 bg-slate-900/20 rounded-xl border border-slate-900";
                grid.appendChild(emptyCell);
            }

            // Days cells
            for (let day = 1; day <= totalDays; day++) {
                const dayStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
              
                // Find logged or scheduled events for this day
                const dayLogs = workouts.filter(w => w.date === dayStr);
                const daySched = schedule.filter(s => s.date === dayStr);

                const cell = document.createElement('div');
                cell.className = "h-20 sm:h-24 bg-slate-900/60 border border-slate-800 hover:border-slate-700 rounded-xl p-1 sm:p-2 flex flex-col justify-between overflow-hidden cursor-pointer transition-all";
                cell.onclick = () => quickAddScheduleForDate(dayStr);

                let badges = '';
                dayLogs.forEach(l => {
                    badges += `<div class="text-[10px] bg-brand-600/80 text-white rounded px-1 py-0.5 truncate flex items-center gap-1 mb-0.5" title="${l.discipline}">
                        ${getDisciplineIcon(l.discipline)} <span class="hidden sm:inline">${l.discipline}</span>
                    </div>`;
                });

                daySched.forEach(s => {
                    badges += `<div class="text-[10px] bg-emerald-600/80 text-white rounded px-1 py-0.5 truncate flex items-center gap-1 mb-0.5" title="Prévu: ${s.title}">
                        <i class="fa-solid fa-clock"></i> <span class="hidden sm:inline">${s.discipline}</span>
                    </div>`;
                });

                cell.innerHTML = `
                    <div class="text-xs font-semibold text-slate-400">${day}</div>
                    <div class="overflow-y-auto max-h-16">${badges}</div>
                `;
                grid.appendChild(cell);
            }

            renderScheduledList();
        }

        function changeMonth(delta) {
            currentCalendarDate.setMonth(currentCalendarDate.getMonth() + delta);
            renderCalendar();
        }

        function quickAddScheduleForDate(dateStr) {
            document.getElementById('sched-date').value = dateStr;
            openScheduleModal();
        }

        function handleSaveSchedule(e) {
            e.preventDefault();
            const newSched = {
                id: 's_' + Date.now(),
                discipline: document.getElementById('sched-discipline').value,
                date: document.getElementById('sched-date').value,
                duration: parseInt(document.getElementById('sched-duration').value || 0),
                title: document.getElementById('sched-title').value || 'Séance programmée',
                completed: false
            };
            schedule.push(newSched);
            saveState();
            closeScheduleModal();
            renderCalendar();
        }

        function renderScheduledList() {
            const container = document.getElementById('scheduled-list');
            container.innerHTML = '';

            if (schedule.length === 0) {
                container.innerHTML = `<p class="text-xs text-slate-500">Aucune séance programmée.</p>`;
                return;
            }

            schedule.sort((a,b) => new Date(a.date) - new Date(b.date)).forEach(s => {
                const item = document.createElement('div');
                item.className = "flex items-center justify-between bg-slate-900/60 p-3 rounded-xl border border-slate-800 text-xs";
                item.innerHTML = `
                    <div class="flex items-center gap-3">
                        <div class="w-8 h-8 rounded-lg bg-emerald-500/10 text-emerald-400 flex items-center justify-center">
                            ${getDisciplineIcon(s.discipline)}
                        </div>
                        <div>
                            <strong class="text-white block">${s.title} (${s.discipline})</strong>
                            <span class="text-slate-400">${formatDate(s.date)} ${s.duration ? '• ' + s.duration + ' min' : ''}</span>
                        </div>
                    </div>
                    <div class="flex items-center gap-2">
                        <button onclick="markScheduleDone('${s.id}')" class="px-2 py-1 bg-emerald-600/20 hover:bg-emerald-600 text-emerald-400 hover:text-white border border-emerald-500/30 rounded-lg transition-colors">
                            Valider comme faite
                        </button>
                        <button onclick="deleteSchedule('${s.id}')" class="text-slate-500 hover:text-rose-400">
                            <i class="fa-solid fa-xmark text-sm"></i>
                        </button>
                    </div>
                `;
                container.appendChild(item);
            });
        }

        function markScheduleDone(id) {
            const item = schedule.find(s => s.id === id);
            if (item) {
                // Open workout form prefilled
                openLogModal();
                document.getElementById('form-discipline').value = item.discipline;
                document.getElementById('form-date').value = item.date;
                document.getElementById('form-duration').value = item.duration || 45;
                document.getElementById('form-notes').value = item.title;
                toggleDisciplineFields();

                // Remove from schedule
                deleteSchedule(id);
            }
        }

        function deleteSchedule(id) {
            schedule = schedule.filter(s => s.id !== id);
            saveState();
            renderCalendar();
        }

        // ================= WEIGHT TRACKER CONTROLLER =================
        function renderWeightSection() {
            document.getElementById('input-target-weight').value = targetWeight;
            
            const latestWeight = weightLogs.length > 0 ? weightLogs[weightLogs.length - 1].weight : 0;
            document.getElementById('weight-last-value').innerText = `${latestWeight} kg`;
            
            const gap = (latestWeight - targetWeight).toFixed(1);
            document.getElementById('weight-gap-value').innerText = `${gap > 0 ? '+' : ''}${gap} kg`;

            // Table Render
            const table = document.getElementById('weight-history-table');
            table.innerHTML = '';

            [...weightLogs].sort((a,b) => new Date(b.date) - new Date(a.date)).forEach(w => {
                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-800/40 transition-colors";
                tr.innerHTML = `
                    <td class="p-3 text-slate-300 font-medium">${formatDate(w.date)}</td>
                    <td class="p-3 text-white font-bold">${w.weight} kg</td>
                    <td class="p-3 text-slate-400 text-xs">${w.note || '-'}</td>
                    <td class="p-3 text-right">
                        <button onclick="deleteWeight('${w.id}')" class="text-slate-500 hover:text-rose-400">
                            <i class="fa-solid fa-trash text-xs"></i>
                        </button>
                    </td>
                `;
                table.appendChild(tr);
            });

            renderWeightChart();
        }

        function updateTargetWeight(e) {
            e.preventDefault();
            targetWeight = parseFloat(document.getElementById('input-target-weight').value);
            saveState();
            renderDashboard();
            renderWeightSection();
        }

        function handleSaveWeight(e) {
            e.preventDefault();
            const newLog = {
                id: 'w_' + Date.now(),
                date: document.getElementById('weight-date').value,
                weight: parseFloat(document.getElementById('weight-val').value),
                note: document.getElementById('weight-note').value

            saveState();
            closeWeightModal();
            renderDashboard();
            renderWeightSection();
        }

        function deleteWeight(id) {
            weightLogs = weightLogs.filter(w => w.id !== id);
            saveState();
            renderDashboard();
            renderWeightSection();
        }

        function renderWeightChart() {
            const ctx = document.getElementById('weightChart');
            if(!ctx) return;

            const sorted = [...weightLogs].sort((a,b) => new Date(a.date) - new Date(a.date));
            const labels = sorted.map(w => formatDate(w.date));
            const data = sorted.map(w => w.weight);

            if (weightChartInstance) weightChartInstance.destroy();

            weightChartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Poids (kg)',
                        data: data,
                        borderColor: '#a855f7',
                        backgroundColor: 'rgba(168, 85, 247, 0.1)',
                        fill: true,
                        tension: 0.3,
                        pointBackgroundColor: '#a855f7'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: {
                            grid: { color: '#334155' },
                            ticks: { color: '#94a3b8' }
                        },
                        x: {
                            grid: { display: false },
                            ticks: { color: '#94a3b8' }
                        }
                    }
                }
            });
        }

        // ================= UTILITY & MODAL HELPERS =================
        function getDisciplineIcon(disc) {
            if (disc === 'Course') return '<i class="fa-solid fa-person-running text-blue-400"></i>';
            if (disc === 'Natation') return '<i class="fa-solid fa-person-swimming text-cyan-400"></i>';
            if (disc === 'Tennis') return '<i class="fa-solid fa-baseball text-amber-400"></i>';
            if (disc === 'Pilates') return '<i class="fa-solid fa-spa text-pink-400"></i>';
            if (disc === 'Musculation') return '<i class="fa-solid fa-dumbbell text-emerald-400"></i>';
            return '<i class="fa-solid fa-bolt text-brand-500"></i>';
        }

        function formatDate(dateStr) {
            if(!dateStr) return '';
            const [y, m, d] = dateStr.split('-');
            return `${d}/${m}/${y}`;
        }
